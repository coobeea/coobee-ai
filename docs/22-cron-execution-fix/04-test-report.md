private normalizeCronExpression(expr: string): string {
  const parts = expr.trim().split(/\s+/);
  if (parts.length === 5) {
    return `0 ${expr}`; // ✅ 5 位 → 6 位
  }
  if (parts.length >= 6) {
    return expr;
  }
  throw new Error(`无效的 cron 表达式格式: ${expr}`);
}
