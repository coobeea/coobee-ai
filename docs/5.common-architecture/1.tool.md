// ==================== 工具类型定义 ====================

/\*_ 工具类型 _/
export enum ToolCategory {
FileSystem = 'file_system', // 文件系统操作
Search = 'search', // 搜索功能
Execute = 'execute', // 执行命令
Web = 'web', // 网络操作
Memory = 'memory', // 记忆系统
Documentation = 'documentation', // 文档
Extension = 'extension' // 扩展
}

/** 工具错误信息 \*/
export interface ToolError {
/** 错误代码 _/
code: string;
/\*\* 错误消息 _/
message: string;
/\*_ 详细信息（可选） _/
details?: any;
}

/** 工具执行结果 \*/
export interface ToolResult {
/** 是否成功 \*/
success: boolean;

/\*_ LLM看到的工具执行结果内容 _/
llmContent?: string;

/\*_ 用户看到的工具执行结果内容 _/
userContent?: string;

/\*_ 失败时的错误信息 _/
error?: ToolError;

/** 执行元数据 \*/
metadata?: {
/** 执行开始时间 _/
startTime?: number;
/\*\* 执行结束时间 _/
endTime?: number;
/** 执行耗时（毫秒） \*/
duration?: number;
/** 结果的 token 数量（估算） _/
tokens?: number;
/\*\* 其他元数据 _/
[key: string]: any;
};
}

/\*\*

- 工具流式更新
-
- @example
- // 进度更新
- yield { type: 'progress', content: '正在处理...', percentage: 50 };
-
- @example
- // 输出内容
- yield { type: 'output', content: '找到 10 个文件' };
  _/
  export interface ToolStreamUpdate {
  /\*\* 更新类型: progress-进度, output-输出内容 _/
  type: 'progress' | 'output';

/\*_ 更新内容 _/
content: string;

/\*_ 进度百分比 (0-100)，仅当 type='progress' 时需要 _/
percentage?: number;
}

/** 工具注册信息: create -> execute \*/
export interface ToolRegistration<TParams = any> extends ChatToolDefinition {
/** 工具类型 \*/
category: ToolCategory;

/\*_ 是否需要用户确认 _/
needUserConfirm: boolean;

/\*_ 是否需要用户输入 _/
needUserInput: boolean;

/\*_ 是否需要用户选择 _/
needUserSelect: boolean;

/\*_ 输出是否为 Markdown _/
isOutputMarkdown: boolean;

/\*_ 创建工具函数 _/
create(params: TParams): ToolFunction<TParams>;
}

/** 工具函数接口 \*/
export interface ToolFunction<TParams = any> {
/** 工具参数类型 \*/
params: TParams;

/\*_ 工具描述 _/
registration: ToolRegistration<TParams>;

/** 获取工具名称 \*/
getName: () => string;
/** 获取工具描述 _/
getDescription: () => string;
/\*\* 获取工具分类 _/
getCategory: () => string;
/** 获取工具参数 \*/
getParams: () => TParams;
/** 获取工具注册信息 \*/
getRegistration: () => ToolRegistration<TParams>;

/\*_ 执行工具函数 _/
execute: (signal: AbortSignal) => AsyncGenerator<ToolStreamUpdate, ToolResult, unknown>;
}
