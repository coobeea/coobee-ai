#### 任务 1.3: 实现 SHA256 计算

**文件**: `workers/brain/brain/asset.py`

```python
import hashlib
import json
from typing import Dict, Any

def compute_asset_id(asset: Dict[str, Any]) -> str:
    """
    计算资产的 SHA256 ID

    参考 EvoMap 规范：
    1. 移除 asset_id 字段
    2. 规范化 JSON（排序 keys）
    3. 计算 SHA256
    """
    # 1. 移除 asset_id
    content = {k: v for k, v in asset.items() if k != 'asset_id'}

    # 2. 规范化 JSON（排序 keys，不转义中文）
    canonical = json.dumps(content, sort_keys=True, ensure_ascii=False)

    # 3. 计算 SHA256
    hash_hex = hashlib.sha256(canonical.encode('utf-8')).hexdigest()

    return f"sha256:{hash_hex}"


def generate_package_id(pattern_id: str, practice_id: str) -> str:
    """
    生成经验包 ID

    格式: pkg_{pattern_hash[:8]}{practice_hash[:8]}
    """
    pattern_hash = pattern_id.replace('sha256:', '')[:8]
    practice_hash = practice_id.replace('sha256:', '')[:8]
    return f"pkg_{pattern_hash}{practice_hash}"
```

#### 任务 1.4: 实现文件系统存储

**文件**: `workers/brain/brain/storage.py`

```python
import os
import json
from pathlib import Path
from typing import Optional, Dict, Any

class FileSystemStore:
    """文件系统存储管理器"""

    def __init__(self, storage_dir: str):
        self.storage_dir = Path(storage_dir)
        self.packages_dir = self.storage_dir / 'packages'
        self.packages_dir.mkdir(parents=True, exist_ok=True)

    def save_package(self, pkg: Dict[str, Any]) -> None:
        """保存经验包"""
        pkg_dir = self.packages_dir / pkg['package_id']
        pkg_dir.mkdir(parents=True, exist_ok=True)

        # 写入主文件
        self._write_json(pkg_dir / 'package.json', pkg)
        self._write_json(pkg_dir / 'pattern.json', pkg['pattern'])
        self._write_json(pkg_dir / 'practice.json', pkg['practice'])

        if 'evolution' in pkg and pkg['evolution']:
            self._write_json(pkg_dir / 'evolution.json', pkg['evolution'])

    def load_package(self, package_id: str) -> Optional[Dict[str, Any]]:
        """加载经验包"""
        pkg_path = self.packages_dir / package_id / 'package.json'

        if not pkg_path.exists():
            return None

        with open(pkg_path, 'r', encoding='utf-8') as f:
            return json.load(f)

    def list_packages(self) -> list[str]:
        """列出所有经验包 ID"""
        if not self.packages_dir.exists():
            return []

        return [d.name for d in self.packages_dir.iterdir() if d.is_dir()]

    def _write_json(self, path: Path, data: Dict[str, Any]) -> None:
        """写入 JSON 文件"""
        with open(path, 'w', encoding='utf-8') as f:
            json.dump(data, f, ensure_ascii=False, indent=2)
```

```python
#!/usr/bin/env python3
import os
import sys
import argparse
from datetime import datetime
from flask import Flask, request, jsonify

# 添加当前目录到 Python path
sys.path.insert(0, os.path.dirname(__file__))

from brain.storage import FileSystemStore
from brain.indexer import IndexManager
from brain.asset import compute_asset_id, generate_package_id

app = Flask(__name__)

# 全局存储（从环境变量获取路径）
storage_dir = None
store = None
indexer = None

def log(level: str, message: str):
    """日志输出"""
    timestamp = datetime.now().strftime('%Y-%m-%d %H:%M:%S')
    print(f"[{timestamp}] [{level}] {message}", flush=True)

@app.route('/health', methods=['GET'])
def health():
    """健康检查端点（WorkerManager 轮询）"""
    return jsonify({'status': 'ok'})

@app.route('/api/brain/publish', methods=['POST'])
def publish():
    """发布经验包"""
    try:
        data = request.get_json()
        message_id = data.get('message_id')
        timestamp = data.get('timestamp')
        payload = data.get('payload', {})

        pattern = payload.get('pattern')
        practice = payload.get('practice')
        evolution = payload.get('evolution')

        if not pattern or not practice:
            return jsonify({
                'success': False,
                'error': {
                    'code': 'BUNDLE_REQUIRED',
                    'message': '必须同时提供 pattern 和 practice'
                }
            }), 400

        # 计算 asset_id
        pattern_id = compute_asset_id(pattern)
        practice_id = compute_asset_id(practice)
        evolution_id = compute_asset_id(evolution) if evolution else None

        # 生成 package_id
        package_id = generate_package_id(pattern_id, practice_id)

        # 构建 Package
        pkg = {
            'package_id': package_id,
            'pattern': {**pattern, 'asset_id': pattern_id},
            'practice': {
                **practice,
                'asset_id': practice_id,
                'pattern_id': pattern_id
            },
            'status': 'candidate',
            'usage_count': 0,
            'created_at': timestamp,
            'updated_at': timestamp
        }

        if evolution:
            pkg['evolution'] = {
                **evolution,
                'asset_id': evolution_id,
                'practice_id': practice_id,
                'patterns_used': [pattern_id]
            }

        # 存储
        store.save_package(pkg)

        # 更新索引
        indexer.add_package(pkg)

        log('INFO', f'Published package: {package_id}')

        return jsonify({
            'success': True,
            'data': {
                'package_id': package_id,
                'pattern_id': pattern_id,
                'practice_id': practice_id,
                'evolution_id': evolution_id,
                'status': 'candidate'
            }
        })

    except Exception as e:
        log('ERROR', f'Publish failed: {str(e)}')
        return jsonify({
            'success': False,
            'error': {
                'code': 'INTERNAL_ERROR',
                'message': str(e)
            }
        }), 500

def main():
    global storage_dir, store, indexer

    parser = argparse.ArgumentParser(description='Brain Worker')
    parser.add_argument('--port', type=int, default=42043, help='Port')
    args = parser.parse_args()

    # 获取存储目录（从环境变量）
    user_home = os.environ.get('USER_HOME')
    if not user_home:
        log('ERROR', 'USER_HOME environment variable not set')
        sys.exit(1)

    storage_dir = os.path.join(user_home, 'brain')
    store = FileSystemStore(storage_dir)
    indexer = IndexManager(storage_dir)

    log('INFO', f'Brain Worker starting on port {args.port}')
    log('INFO', f'Storage directory: {storage_dir}')

    app.run(host='127.0.0.1', port=args.port, debug=False)

if __name__ == '__main__':
    main()
```

**文件**: `workers/brain/brain/indexer.py`

```typescript
import { promises as fs } from 'fs';
import path from 'path';

export class IndexManager {
  constructor(private storageDir: string) {}

  async addPackage(pkg: Package): Promise<void> {
    // 按信号索引
    for (const signal of pkg.pattern.signals) {
      await this.appendIndex('by-signal', {
        signal,
        package_id: pkg.package_id,
        timestamp: pkg.created_at
      });
    }

    // 按类别索引
    await this.appendIndex('by-category', {
      category: pkg.pattern.category,
      package_id: pkg.package_id,
      timestamp: pkg.created_at
    });

    // 按状态索引
    await this.appendIndex('by-status', {
      status: pkg.status,
      package_id: pkg.package_id,
      timestamp: pkg.created_at
    });
  }

  async findBySignals(signals: string[]): Promise<string[]> {
    const indexPath = path.join(this.storageDir, 'index', 'by-signal.jsonl');

    const content = await fs.readFile(indexPath, 'utf-8');
    const lines = content.trim().split('\n');

    const matches = new Set<string>();
    for (const line of lines) {
      const entry = JSON.parse(line);
      if (signals.includes(entry.signal)) {
        matches.add(entry.package_id);
      }
    }

    return Array.from(matches);
  }

  private async appendIndex(indexName: string, entry: Record<string, any>): Promise<void> {
    const indexPath = path.join(this.storageDir, 'index', `${indexName}.jsonl`);

    await fs.mkdir(path.dirname(indexPath), { recursive: true });
    await fs.appendFile(indexPath, JSON.stringify(entry) + '\n', 'utf-8');
  }
}
```

```typescript
import type { Context } from 'koa';
import { AssetCompute } from '../../storage/AssetCompute';
import { FileSystemStore } from '../../storage/FileSystemStore';

export async function publishHandler(ctx: Context) {
  const { message_id, timestamp, payload } = ctx.request.body;

  // 1. 验证必需字段
  if (!payload.pattern || !payload.practice) {
    ctx.status = 400;
    ctx.body = {
      success: false,
      error: {
        code: 'BUNDLE_REQUIRED',
        message: '必须同时提供 pattern 和 practice'
      }
    };
    return;
  }

  // 2. 计算 asset_id
  const patternId = AssetCompute.computeAssetId(payload.pattern);
  const practiceId = AssetCompute.computeAssetId(payload.practice);
  const evolutionId = payload.evolution ? AssetCompute.computeAssetId(payload.evolution) : null;

  // 3. 生成 package_id
  const packageId = `pkg_${patternId.slice(7, 15)}${practiceId.slice(7, 15)}`;

  // 4. 构建 Package
  const pkg: Package = {
    package_id: packageId,
    pattern: { ...payload.pattern, asset_id: patternId },
    practice: {
      ...payload.practice,
      asset_id: practiceId,
      pattern_id: patternId
    },
    evolution: evolutionId
      ? {
          ...payload.evolution,
          asset_id: evolutionId,
          practice_id: practiceId,
          patterns_used: [patternId]
        }
      : undefined,
    status: 'candidate',
    usage_count: 0,
    created_at: timestamp,
    updated_at: timestamp
  };

  // 5. 存储
  const store = new FileSystemStore(storageDir);
  await store.savePackage(pkg);

  // 6. 返回
  ctx.status = 200;
  ctx.body = {
    success: true,
    data: {
      package_id: packageId,
      pattern_id: patternId,
      practice_id: practiceId,
      evolution_id: evolutionId,
      status: 'candidate'
    }
  };
}
```

```typescript
export async function searchHandler(ctx: Context) {
  const { payload } = ctx.request.body;
  const { signals, category, status, limit = 10 } = payload;

  const indexManager = new IndexManager(storageDir);
  const store = new FileSystemStore(storageDir);

  // 1. 根据 signals 找到 package_id
  let packageIds: string[] = [];
  if (signals && signals.length > 0) {
    packageIds = await indexManager.findBySignals(signals);
  } else {
    // 返回所有
    packageIds = await indexManager.getAllPackageIds();
  }

  // 2. 加载 Package 并过滤
  const packages = await Promise.all(packageIds.map((id) => store.loadPackage(id)));

  let filtered = packages.filter(Boolean);

  if (category) {
    filtered = filtered.filter((pkg) => pkg.pattern.category === category);
  }

  if (status) {
    filtered = filtered.filter((pkg) => pkg.status === status);
  }

  // 3. 限制数量
  const limited = filtered.slice(0, limit);

  // 4. 返回（只返回摘要，不含完整 content）
  ctx.status = 200;
  ctx.body = {
    success: true,
    data: {
      packages: limited.map((pkg) => ({
        package_id: pkg.package_id,
        pattern: {
          name: pkg.pattern.name,
          summary: pkg.pattern.summary
        },
        practice: {
          name: pkg.practice.name,
          summary: pkg.practice.summary,
          confidence: pkg.practice.confidence
        },
        status: pkg.status,
        usage_count: pkg.usage_count
      })),
      total: filtered.length
    }
  };
}
```
