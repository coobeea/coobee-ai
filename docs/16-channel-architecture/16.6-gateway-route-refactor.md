# 16.6 - Gateway HTTP 路由调整

> **任务 ID**: Task-6  
> **优先级**: P1  
> **依赖**: Task-5（Discussion Channel 迁移）  
> **预计工期**: 0.2 天

---

## 1. 需求描述

### 1.1 用户需求

**背景**：

- 当前 Gateway HTTP 路由直接操作 `DiscussionRoom` 类
- 讨论室已改造为 ChannelPlugin，路由需要适配
- 需要通过 ChannelPlugin 接口调用，而非直接实例化 DiscussionRoom

**核心需求**：

1. 重构 `/gateway/discussion/*` 路由
2. 通过 ChannelManager 获取 ChannelPlugin
3. 调用 Plugin 的 inbound/outbound 方法
4. 保持 API 签名不变（前端无感知）

**期望效果**：

- 路由代码更简洁
- 逻辑更清晰（Gateway → Plugin → Runtime）
- 易于扩展（未来可支持其他 Channel）

---

### 1.2 技术约束

1. **API 不变**：
   - 路径不变（`/gateway/discussion/*`）
   - 请求/响应格式不变
   - HTTP 方法不变（POST/GET/DELETE）

2. **错误处理**：
   - Channel 不存在 → 404
   - Agent 执行失败 → 500
   - 参数校验失败 → 400

3. **性能要求**：
   - 路由响应延迟 < 100ms（不含 LLM 时间）
   - 支持并发请求

---

## 2. 解决方案

### 2.1 设计思路

**重构策略**：从直接调用改为通过 ChannelPlugin 调用

```
【旧路由】
Gateway HTTP → new DiscussionRoom() → agentExecutor

【新路由】
Gateway HTTP → ChannelManager.getChannelPlugin('discussion')
            → plugin.inbound.handleMessage()
            → ChannelRuntime.executeAgent()
```

**核心变化**：

| 旧方式                    | 新方式                          | 优势 |
| ------------------------- | ------------------------------- | ---- |
| 直接实例化 DiscussionRoom | 通过 ChannelManager 获取 Plugin | 解耦 |
| 手动调用 agentExecutor    | Plugin 自动调度                 | 统一 |
| 分散的错误处理            | 统一错误处理                    | 健壮 |

## 3. 详细设计

### 3.1 重构路由代码

#### 3.1.1 创建讨论室（保持不变）

```typescript
// src/main/gateway/http/discussion.ts

import { ChannelManager } from '@main/channels/ChannelManager';
import { DiscussionStore } from '@main/ai/discussion/DiscussionStore';
import { DiscussionRoom } from '@main/ai/discussion/DiscussionRoom';

/**
 * 创建讨论室
 * POST /gateway/discussion/sessions
 *
 * Request Body:
 * {
 *   topic: string;
 *   participants: Array<{ agentId: string; roleName: string }>;
 *   turnStrategy?: 'round-robin' | 'free-for-all';
 * }
 */
router.post('/discussion/sessions', async (ctx: Context) => {
  try {
    const {
      topic,
      participants,
      turnStrategy = 'round-robin'
    } = ctx.request.body as {
      topic: string;
      participants: Array<{ agentId: string; roleName: string; active?: boolean }>;
      turnStrategy?: 'round-robin' | 'free-for-all';
    };

    // 1. 参数校验
    if (!topic || !participants || participants.length < 2) {
      ctx.status = 400;
      ctx.body = { error: 'Invalid parameters' };
      return;
    }

    // 2. 创建 session
    const store = await DiscussionStore.getInstance();
    const session = await store.create({
      topic,
      participants: participants.map((p) => ({
        ...p,
        active: p.active ?? true
      })),
      turnStrategy,
      status: 'pending',
      messages: [],
      createdAt: Date.now(),
      updatedAt: Date.now()
    });

    ctx.body = { success: true, session };
  } catch (err) {
    ctx.status = 500;
    ctx.body = { error: err instanceof Error ? err.message : 'Internal server error' };
  }
});
```

#### 3.1.2 开始讨论（重构）

```typescript
/**
 * 开始讨论
 * POST /gateway/discussion/sessions/:id/start
 *
 * 触发第一个 Agent 发言
 */
router.post('/discussion/sessions/:id/start', async (ctx: Context) => {
  try {
    const { id } = ctx.params;

    // 1. 获取 Discussion Channel Plugin
    const manager = ChannelManager.getInstance();
    const plugin = manager.getChannelPlugin('discussion');

    if (!plugin) {
      ctx.status = 404;
      ctx.body = { error: 'Discussion channel not available' };
      return;
    }

    if (!plugin.inbound) {
      ctx.status = 500;
      ctx.body = { error: 'Discussion channel does not support inbound messages' };
      return;
    }

    // 2. 加载讨论室
    const store = await DiscussionStore.getInstance();
    const session = await store.get(id);

    if (!session) {
      ctx.status = 404;
      ctx.body = { error: 'Discussion session not found' };
      return;
    }

    // 3. 更新状态为 active
    await store.update(id, { status: 'active' });

    // 4. 添加系统消息
    await store.addMessage(id, {
      participant: 'System',
      content: `Discussion started. Topic: ${session.topic}`,
      timestamp: Date.now()
    });

    // 5. 选择第一个发言者
    const firstSpeaker = session.participants[0];

    // 6. 触发 Plugin 的 inbound.handleMessage
    await plugin.inbound.handleMessage({
      peer: id,
      from: firstSpeaker.agentId,
      text: `You are ${firstSpeaker.roleName}. Please start the discussion on: ${session.topic}`,
      context: {
        channel: 'discussion',
        roomId: id,
        role: firstSpeaker.roleName,
        topic: session.topic,
        recentMessages: []
      }
    });

    ctx.body = { success: true };
  } catch (err) {
    ctx.status = 500;
    ctx.body = { error: err instanceof Error ? err.message : 'Internal server error' };
  }
});
```

#### 3.1.5 手动触发下一轮（可选）

```typescript
/**
 * 下一轮（手动触发）
 * POST /gateway/discussion/sessions/:id/next-turn
 */
router.post('/discussion/sessions/:id/next-turn', async (ctx: Context) => {
  try {
    const { id } = ctx.params;

    const manager = ChannelManager.getInstance();
    const plugin = manager.getChannelPlugin('discussion');

    if (!plugin || !plugin.inbound) {
      ctx.status = 404;
      ctx.body = { error: 'Discussion channel not available' };
      return;
    }

    const store = await DiscussionStore.getInstance();
    const session = await store.get(id);

    if (!session) {
      ctx.status = 404;
      ctx.body = { error: 'Discussion session not found' };
      return;
    }

    // 选择下一个发言者（轮询策略）
    const lastMessage = session.messages[session.messages.length - 1];
    const currentIndex = session.participants.findIndex((p) => p.agentId === lastMessage?.participant);
    const nextIndex = (currentIndex + 1) % session.participants.length;
    const nextSpeaker = session.participants[nextIndex];

    // 触发 Plugin
    await plugin.inbound.handleMessage({
      peer: id,
      from: nextSpeaker.agentId,
      text: `Continue the discussion. Previous messages: ${session.messages
        .slice(-3)
        .map((m) => m.content)
        .join('\n')}`,
      context: {
        channel: 'discussion',
        roomId: id,
        role: nextSpeaker.roleName,
        topic: session.topic,
        recentMessages: session.messages.slice(-5).map((m) => ({
          sender: m.participant,
          content: m.content
        }))
      }
    });

    ctx.body = { success: true };
  } catch (err) {
    ctx.status = 500;
    ctx.body = { error: err instanceof Error ? err.message : 'Internal server error' };
  }
});
```

### 3.2 错误处理统一

```typescript
// 统一错误处理中间件

/**
 * 讨论室错误处理器
 */
function discussionErrorHandler() {
  return async (ctx: Context, next: () => Promise<void>) => {
    try {
      await next();
    } catch (err) {
      if (err instanceof Error) {
        // 根据错误类型返回不同状态码
        if (err.message.includes('not found')) {
          ctx.status = 404;
          ctx.body = { error: err.message };
        } else if (err.message.includes('Invalid')) {
          ctx.status = 400;
          ctx.body = { error: err.message };
        } else {
          ctx.status = 500;
          ctx.body = { error: 'Internal server error' };
        }
      } else {
        ctx.status = 500;
        ctx.body = { error: 'Unknown error' };
      }
    }
  };
}

// 应用到所有讨论室路由
router.use('/discussion', discussionErrorHandler());
```
